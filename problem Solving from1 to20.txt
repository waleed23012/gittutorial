int random_number_from_to(int from, int to) {
	int random = rand() % (to - from + 1) + from;
	return random;  
}

void fill_array(int arr[3][3],int from,int to) {
	for (int i = 0; i < 3; i++) {
		for (int j = 0; j < 3; j++) {
			arr[i][j] = random_number_from_to(from, to);
		}
	}
}

void print_array(int arr[3][3],int row,int col) {
	for (int i = 0; i < row; i++) {
		for (int j = 0; j < col; j++) {
			cout << setfill('0') << setw(2) << arr[i][j] << " ";
		}
		cout << endl;
	}
}

		}

}

int sumation_element_in_matrex(int arr[3][3], int row, int col) {
	int counter = 0;
	for (int i = 0; i < row; i++) {
		for (int j = 0; j < col; j++) {
			counter += arr[i][j];
		}
	}
	return counter;
}

void if_sparse_matrex(int arr[3][3]) {
	int counter_zeros = 0;
	int counter_non_zeros = 0;
	for (int i = 0; i < 3; i++) {
		for (int j = 0; j < 3; j++) {
			if (arr[i][j] == 0) {
				counter_zeros++;
		}
			else {
				counter_non_zeros++;
			}
		}
	}
	if (counter_non_zeros > counter_zeros) {
		cout << "the matrex is not sparse";
	}
	else {
		cout << "the matrex is  sparse";

	}
}
int min_number_in_matrex(int arr[3][3]) {
	int min_num = arr[0][0];
	for (int i = 0; i < 3; i++) {
		for (int j = 0; j < 3; j++) {
			if (min_num > arr[i][j]) {
				min_num = arr[i][j];
			}
		}
	}
	return min_num;
}

			if (max_num < arr[i][j]) {
				max_num = arr[i][j];
			}
		}
	}
	return max_num;
}
